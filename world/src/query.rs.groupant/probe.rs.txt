// Compile as a standalone std-only builder; run from repository root with external output dir.
// Extracts unchanged attention and coverage functions, without substituting their bodies.
// Stubs expose only dependency interfaces; full World/parser/lexicon integration is not claimed.
use std::{fs,path::Path,process::Command};
fn extract(src:&str,from:&str,to:&str)->String {
 let a=src.find(from).expect("start marker"); let b=a+src[a..].find(to).expect("end marker");
 src[a..b].to_owned()
}
fn run(dir:&Path,name:&str,body:&str,pass:bool){
 let source=dir.join(format!("{name}.rs"));let binary=dir.join(name);
 fs::write(&source,body).unwrap();
 assert!(Command::new("rustc").args(["--edition=2024","--crate-name",name]).arg(&source).arg("-o").arg(&binary).status().unwrap().success());
 let out=Command::new(binary).output().unwrap();
 assert_eq!(out.status.success(),pass,"{name}: {}",String::from_utf8_lossy(&out.stderr));
 if pass { print!("{}",String::from_utf8_lossy(&out.stdout)); } else {println!("{name}: rejected");}
}

use std::{env,fs,path::Path,process::Command,collections::BTreeMap};
fn main(){let a:Vec<_>=env::args().collect();let dir=Path::new(&a[1]);let mut values:BTreeMap<(u32,u32,String),Vec<f64>>=BTreeMap::new();let mut ids=BTreeMap::new();
for round in 0..3 {let tags=if round%2==0 {["before","after"]}else{["after","before"]};for tag in tags {let out=Command::new(dir.join(format!("bench-{tag}"))).output().unwrap();fs::write(dir.join(format!("bench-{tag}-{round}.tsv")),&out.stdout).unwrap();fs::write(dir.join(format!("bench-{tag}-{round}.log")),&out.stderr).unwrap();assert!(out.status.success(),"benchmark failed: {tag}");let s=String::from_utf8(out.stdout).unwrap();for l in s.lines().skip(1){let c:Vec<_>=l.split('\t').collect();assert_eq!(c.len(),5);let key=(c[0].parse::<u32>().unwrap(),c[1].parse::<u32>().unwrap());assert_eq!(c[2],"60");if let Some(prev)=ids.insert(key,c[4].to_string()){assert_eq!(prev,c[4]);}values.entry((key.0,key.1,tag.to_string())).or_default().push(c[3].parse().unwrap());}println!("finished {tag} round {round}");}}
let mut report=String::from("| N | changed IDs | before ms | after ms | delta |\n|---:|---:|---:|---:|---:|\n");for n in [128,2048]{for count in [0,1,n]{let median=|tag:&str|{let mut v=values[&(n,count,tag.into())].clone();assert_eq!(v.len(),3);v.sort_by(f64::total_cmp);v[1]};let b=median("before");let c=median("after");report+=&format!("| {n} | {count} | {b:.3} | {c:.3} | {:+.1}% |\n",100.*(c/b-1.));}}fs::write(dir.join("benchmark-summary.md"),&report).unwrap();print!("{report}");}

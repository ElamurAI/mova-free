Proposed oracles are not executed evidence.\n");
 let entries:Vec<Value>=models.iter().map(|(name,typ,body)|{cat+=&format!("\n## {name}\n\nType: {typ}.\n\n{body}\n");json!({"id":name,"type":typ,"status":"partial_source_derived_model","path":"catalog.md","section":name,"subject_revision":hash,"domain":"Finite admitted calls and filesystem states; exact specialization in catalog","observation_scope":"One call or finite publication trace","operator":body,"approximation_semantics":"Conditional source relation; candidate inner complement bound or ambiguous converse","assumptions":common,"evidence":if fragment{"../tests.md"}else{"tests.md"}})}).collect();fs::write(root.join("catalog.md"),cat).unwrap();
 let mut m=json!({"schema":1,"kind":if fragment{"groupant_fragment"}else{"groupant"},"id":id,"status":"partial_source_derived_instance","source":{"path":if fragment{"../../store.rs"}else{"../store.rs"},"sha256":hash},"components":{"ast":{"status":"partial","format":"rust_syntax_projection","path":if fragment{"../syntax.json"}else{"syntax.json"}},"approximant":{"status":"partial","path":"descriptant.md","section":"Forward"},"invariant":{"status":"partial","path":"descriptant.md","section":"Invariant"},"descriptant":{"status":"present","path":"descriptant.md"},"cost":{"status":"symbolic_unmeasured","path":"descriptant.md","section":"Cost"},"obernants":entries,"tests":{"status":"bounded_evidence","path":if fragment{"../tests.md"}else{"tests.md"}}}});if let Some(c)=coverage{m["coverage"]=c}fs::write(root.join("manifest.json"),serde_json::to_string_pretty(&m).unwrap()).unwrap();
}

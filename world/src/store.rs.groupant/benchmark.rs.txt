fn main()->Result<()> {
 let base=std::env::temp_dir().join(format!("mova-sampled-carry-bench-{}",std::process::id()));
 std::fs::create_dir(&base)?;
 struct Cleanup(PathBuf);impl Drop for Cleanup{fn drop(&mut self){let _=std::fs::remove_dir_all(&self.0);}}let _cleanup=Cleanup(base.clone());
 println!("n\tdelta\tcalls\tmilliseconds\tids");
 for n in [128u32,2048] {for count in [0u32,1,n] {
  let dir=base.join(format!("{n}-{count}"));std::fs::create_dir(&dir)?;
  let mut meta=Vec::new();for i in 0..n {meta.extend(0u32.to_le_bytes());meta.extend((i/16).to_le_bytes());meta.extend(0u16.to_le_bytes());meta.extend(1u16.to_le_bytes());meta.extend((i as u64).to_le_bytes());meta.extend(1u32.to_le_bytes());}
  std::fs::write(dir.join("meta.bin"),meta)?;std::fs::write(dir.join("versions.tsv"),b"a\t-\nb\ta\n")?;
  let mut dict=vec![];for i in 0..count {dict.extend(i.to_le_bytes());dict.extend(0u64.to_le_bytes());dict.extend(0u32.to_le_bytes());}std::fs::write(dir.join("dict-b.bin"),dict)?;
  let mut postings=vec![];for i in 0..n {postings.extend(i.to_le_bytes());postings.push(1u8);}for h in ["a","b"] {std::fs::write(dir.join(format!("features-{h}.tsv")),format!("pred\t0\t{n}\n"))?;std::fs::write(dir.join(format!("index-{h}.bin")),&postings)?;}
  let q="pred>=1 sample=32/7";let expected=select(&dir,q,"b")?;assert_eq!(expected.len(),32);set_save(&dir,"bench",q,"a")?;
  let path=dir.join("sets/bench.tsv");let original=std::fs::read(&path)?;
  let saved=format!("# query\t{q}\n# model\tb\n{}\n",expected.iter().map(u32::to_string).collect::<Vec<_>>().join("\n"));
  let mut elapsed=std::time::Duration::ZERO;
  for k in 0..68 {std::fs::write(&path,&original)?;let start=std::time::Instant::now();let got=set_load(&dir,"bench","b")?;let time=start.elapsed();if k>=8 {elapsed+=time;}assert_eq!(got,expected);assert_eq!(std::fs::read(&path)?,saved.as_bytes());}
  println!("{n}\t{count}\t60\t{:.6}\t{}",elapsed.as_secs_f64()*1000.0,expected.iter().map(u32::to_string).collect::<Vec<_>>().join(","));
 }}Ok(())
}

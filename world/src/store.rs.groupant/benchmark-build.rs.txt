use std::{env,fs,path::PathBuf,process::Command};
// Arguments: run directory, before/after label, source-bound probe.txt, anyhow rlib.
fn main(){let a:Vec<_>=env::args().collect();assert_eq!(a.len(),5);let run=PathBuf::from(&a[1]);let tag=&a[2];let source=fs::read_to_string(&a[3]).unwrap();let at=source.rfind("fn main()").expect("probe main");let body=fs::read_to_string(run.join("benchmark.rs.txt")).unwrap();let p=run.join(format!("bench-{tag}.rs"));fs::write(&p,format!("{}{}",&source[..at],body)).unwrap();let lib=PathBuf::from(&a[4]);let deps=lib.parent().unwrap();let o=Command::new("rustc").args(["--edition=2021","-C","opt-level=3","-L",&format!("dependency={}",deps.display()),"--extern",&format!("anyhow={}",lib.display())]).arg(&p).arg("-o").arg(run.join(format!("bench-{tag}"))).output().unwrap();fs::write(run.join(format!("bench-{tag}-build.log")),&o.stderr).unwrap();assert!(o.status.success(),"{}",String::from_utf8_lossy(&o.stderr));println!("built {tag}");}

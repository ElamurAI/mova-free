use serde_json::Value;use std::{env,fs,path::Path,process::Command,collections::BTreeMap};
fn main(){let a:Vec<_>=env::args().collect();let run=Path::new(&a[1]);let root=Path::new(&a[2]);let src=fs::read_to_string(root.join("world/src/store.rs")).unwrap();let syn:Value=serde_json::from_slice(&fs::read(run.join("store-syntax.json")).unwrap()).unwrap();let mut body=String::from("use en::gram::{UPos,Rel};\n");let names=["decode","decode_tag","same_stored_tree"];let mut found=vec![];for n in syn["nodes"].as_array().unwrap(){if n["category"]!="item"{continue}let s=&src[n["start"].as_u64().unwrap() as usize..n["end"].as_u64().unwrap() as usize];for name in names{if s.contains(&format!("fn {name}(")){assert!(!found.contains(&name));found.push(name);body.push_str(s);body.push('\n');}}}assert_eq!(found.len(),3);body.push_str(&fs::read_to_string(run.join("delta-benchmark.rs.txt")).unwrap());fs::write(run.join("bench.rs"),body).unwrap();
let lib=Path::new(&a[3]);let deps=lib.parent().unwrap().display().to_string();let en_rlib=lib.display().to_string();let o=Command::new("rustc").args(["--edition=2021","-C","opt-level=3","-L",&format!("dependency={deps}"),"--extern",&format!("en={en_rlib}")]).arg(run.join("bench.rs")).arg("-o").arg(run.join("bench")).output().unwrap();fs::write(run.join("bench-build.log"),&o.stderr).unwrap();assert!(o.status.success(),"{}",String::from_utf8_lossy(&o.stderr));
let mut values:BTreeMap<(usize,String,String),Vec<f64>>=BTreeMap::new();let mut answers=BTreeMap::new();for round in 0..3{for tag in if round%2==0{["reference","candidate"]}else{["candidate","reference"]}{let o=Command::new(run.join("bench")).arg(tag).output().unwrap();assert!(o.status.success(),"{}",String::from_utf8_lossy(&o.stderr));fs::write(run.join(format!("{tag}-{round}.tsv")),&o.stdout).unwrap();for l in String::from_utf8(o.stdout).unwrap().lines().skip(1){let c:Vec<_>=l.split('\t').collect();assert_eq!(c.len(),5);let key=(c[0].parse::<usize>().unwrap(),c[1].to_string());if let Some(old)=answers.insert(key.clone(),c[4].to_string()){assert_eq!(old,c[4]);}values.entry((key.0,key.1,tag.into())).or_default().push(c[3].parse().unwrap());}println!("finished {tag} {round}");}}
let mut text=String::from("| tokens | case | reference ms | fused ms | delta |\n|---:|---|---:|---:|---:|\n");for n in [8,128]{for case in ["ascii","unicode","late-change"]{let med=|tag:&str|{let mut v=values[&(n,case.into(),tag.into())].clone();assert_eq!(v.len(),3);v.sort_by(f64::total_cmp);v[1]};let r=med("reference");let c=med("candidate");text+=&format!("| {n} | {case} | {r:.3} | {c:.3} | {:+.1}% |\n",100.*(c/r-1.));}}fs::write(run.join("benchmark-summary.md"),&text).unwrap();print!("{text}");}

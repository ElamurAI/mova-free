fn emit(path:&str,id:&str,bounds:Option<(usize,usize)>,models:&[Model],forward:&str,inv:&str,cost:&str,children:&[&str],src:&str,ast:&Value){
 let rel=if path.is_empty(){String::new()}else{format!("{path}.groupant")};
 let dir=Path::new(ROOT).join(&rel);fs::create_dir_all(&dir).unwrap();
 let depth=if path.is_empty(){0}else{path.matches('/').count()+1};let up="../".repeat(depth);
 let mut text=format!("# {id}\n\nSubject: world/src/span.rs, whole-file SHA-256 in manifest, baseline 4279da541. Domain: finite acyclic valid trees with consistent child lists, valid indices and terminating dependencies. Observation: a single call return, not semantic truth or reachability. Hold Tree, ans::lemmas and types::FEELING dependencies fixed; parser/model loading and external caller wiring are gaps. All entries are source-derived partial models. Exact-target definitions are not executable inverses; finite tests cannot establish universal correspondence.\n\n## forward\n\n{forward}\n\n## invariant\n\n{inv}\n\n## cost\n\n{cost}\n\n## epoch-memory\n\n2026-10-29: initial source-bound epoch. Preserve Num bypass, global Feel selection, early-return cleanup suppression, and verb_kids comment mismatch: code expands Xcomp only and checks no shared subject. No source changes or task-accuracy claim. Partial syntax has no semantic analysis or macro expansion.\n\n## checks\n\nStructural inventory checks hashes, references, types and node bounds only. Proposed oracle: enumerate small annotated acyclic trees and compare literal expected returns, including None and labels; mutate priority or echo inequality to obtain a red control. Any executed bounded evidence is recorded separately in root evidence.md; catalog examples remain proposed until recorded there.\n\n## composition\n\n");
 text.push_str(if children.is_empty(){"Leaf scope shares immutable tree/question/helper contracts. Overlap with parent is not extra runtime work. Dependencies outside range remain opaque.\n"}else{"Children share immutable Tree and question state. Routing yields Want, predicate selection yields p, dispatch yields candidate indices, normalization yields text; Num bypasses normalization. Guarded branches compose as a union, sequential stages via existential intermediates. Reverse keeps all compatible intermediates; complement is this node's U minus composed R, not union of child complements. Why/clauses overlap extraction. minus_question is separate API and caller wiring is outside file. Do/Clause delegation falls through after After None but most direct branch returns stop on clean None.\n"});
 let entries:Vec<Value>=models.iter().map(|(name,ty,body)|{
 text.push_str(&format!("\n## {name}\n\n{body}\n\nAssumptions/evidence/status: common domain above; source-derived partial relation at bound revision. Approximation and held-fixed inputs are stated in the relation. Oracle: finite literal fixtures exercising both membership and nonmembership against the actual function, preserving ambiguous preimages. No linguistic correctness or universal proof inferred.\n"));
 json!({"id":format!("{id}-{name}"),"type":ty,"path":"model.md","section":name,"status":"partial_source_model"})
 }).collect();
 fs::write(dir.join("model.md"),text).unwrap();
 let mut c=json!({"ast":{"status":"partial","format":"rust_syntax_projection","path":format!("{up}syntax.json")},"obernants":entries});
 for(key,section)in[("approximant","forward"),("invariant","invariant"),("cost","cost"),("descriptant","epoch-memory"),("tests","checks")]{
 c[key]=json!({"path":"model.md","section":section,"status":"partial_source_model"});
 }
 if !children.is_empty(){c["composition"]=json!({"path":"model.md","section":"composition","status":"partial_source_model"});}
 let mut m=json!({"schema":1,"kind":if depth==0{"groupant"}else{"groupant_fragment"},"id":id,"status":"partial_source_bound_model","source":{"path":format!("{up}../span.rs"),"sha256":ast["source"]["sha256"]},"components":c});
 if let Some((lo,hi))=bounds{
 let nodes:Vec<&Value>=ast["nodes"].as_array().unwrap().iter().filter(|n|{
 let a=n["start"].as_u64().unwrap()as usize;let b=n["end"].as_u64().unwrap()as usize;
 a>=lo&&b<=hi&&b>a&&(n["category"]=="item"||n["category"]=="expression")
 }).collect();
 let a=nodes.iter().map(|n|n["start"].as_u64().unwrap()).min().unwrap();
 let b=nodes.iter().map(|n|n["end"].as_u64().unwrap()).max().unwrap();
 let ids:Vec<_>=nodes.iter().filter(|n|n["start"].as_u64()==Some(a)||n["end"].as_u64()==Some(b)).map(|n|n["id"].clone()).collect();
 m["coverage"]=json!({"description":format!("Source scope beginning {}; selected node envelope, partial semantic coverage.",src[lo..hi].lines().next().unwrap()),"byte_range":{"start":a,"end":b},"syntax_nodes":{"path":format!("{up}syntax.json"),"node_ids":ids}});
 }
 if !children.is_empty(){m["children"]=json!(children.iter().map(|name|json!({"id":format!("answer-span-{name}"),"path":format!("{name}.groupant")})).collect::<Vec<_>>());}
 fs::write(dir.join("manifest.json"),serde_json::to_string_pretty(&m).unwrap()+"\n").unwrap();
}

### vmm-ctl-031 | recursion | recursive factorial, fibonacci and gcd | src: exact formula

printf('%d\n', mygcd(1071, 462));

function r = fact(n)
  if n <= 1
    r = 1;
  else
    r = n * fact(n - 1);
  end
end
function r = fib(n)
  if n < 2
    r = n;
  else
    r = fib(n - 1) + fib(n - 2);
  end
end
function g = mygcd(a, b)
  if b == 0
    g = a;
  else
    g = mygcd(b, mod(a, b));
  end
end
